cmake_minimum_required(VERSION 3.21)
project(SR3LabStandalone C)
if(NOT CMAKE_SIZEOF_VOID_P EQUAL 4)
  message(FATAL_ERROR "Rally.exe requires an x86 (32-bit) build. Use -A Win32 with Visual Studio.")
endif()
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
if(NOT SDL_SOURCE OR NOT EXISTS "${SDL_SOURCE}/CMakeLists.txt")
  message(FATAL_ERROR "Set SDL_SOURCE to the pinned SDL 3.4.16 source cache. The default build.py uses GCC directly.")
endif()
set(SDL_SHARED OFF CACHE BOOL "" FORCE)
set(SDL_STATIC ON CACHE BOOL "" FORCE)
set(SDL_STATIC_VCRT ON CACHE BOOL "" FORCE)
set(SDL_TEST_LIBRARY OFF CACHE BOOL "" FORCE)
set(SDL_TESTS OFF CACHE BOOL "" FORCE)
set(SDL_EXAMPLES OFF CACHE BOOL "" FORCE)
foreach(feature AUDIO VIDEO RENDER CAMERA GPU OPENGL OPENGLES VULKAN OFFSCREEN)
  set(SDL_${feature} OFF CACHE BOOL "" FORCE)
endforeach()
add_subdirectory("${SDL_SOURCE}" sdl EXCLUDE_FROM_ALL)
add_library(sr3runtime SHARED runtime.c runtime.def settings.c patch_loader.c capture.c)
target_link_libraries(sr3runtime PRIVATE SDL3::SDL3-static user32)
target_compile_definitions(sr3runtime PRIVATE _CRT_SECURE_NO_WARNINGS)
if(MSVC)
  target_compile_options(sr3runtime PRIVATE /W4)
  target_link_options(sr3runtime PRIVATE /Brepro)
endif()
set_target_properties(sr3runtime PROPERTIES PREFIX "" OUTPUT_NAME "dinput8" ARCHIVE_OUTPUT_NAME "sr3runtime")
enable_testing()
add_executable(test_input test_input.c)
add_test(NAME input_report COMMAND test_input)
add_executable(sr3launcher WIN32 launcher.c launch.c watch_launcher.c settings.c)
target_compile_definitions(sr3launcher PRIVATE _CRT_SECURE_NO_WARNINGS)
set_target_properties(sr3launcher PROPERTIES OUTPUT_NAME "launch")
add_executable(test_settings test_settings.c settings.c)
target_compile_definitions(test_settings PRIVATE _CRT_SECURE_NO_WARNINGS)
add_test(NAME settings_roundtrip COMMAND test_settings)
add_executable(test_capture test_capture.c)
add_test(NAME capture_mock COMMAND test_capture)
